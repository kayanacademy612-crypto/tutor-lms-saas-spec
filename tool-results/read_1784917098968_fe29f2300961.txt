    11→    11→// CollectionSchema pairs a collection name with its JSON Schema validator.
    12→    12→type CollectionSchema struct {
    13→    13→        Collection string
    14→    14→        Schema     bson.M
    15→    15→}
    16→    16→
    17→    17→// AllSchemas returns the JSON Schema validators for all 15 validated collections.
    18→    18→func AllSchemas() []CollectionSchema {
    19→    19→        return []CollectionSchema{
    20→    20→                usersSchema(),
    21→    21→                tenantsSchema(),
    22→    22→                tenantMembershipsSchema(),
    23→    23→                invitationsSchema(),

    35→    35→                eventDefinitionsSchema(),
    36→    36→        }
    37→    37→}
    38→    38→
    39→    39→// EnsureSchemaValidation applies JSON Schema validators to all validated
    40→    40→// collections using collMod with moderate validation level.
    41→    41→func (m *MongoDB) EnsureSchemaValidation() {
    42→    42→        ctx, cancel := context.WithTimeout(context.Background(), 60*time.Second)
    43→    43→        defer cancel()
    44→    44→
    45→    45→        for _, cs := range AllSchemas() {
    46→    46→                // Ensure the collection exists (ignore "already exists" errors).
    47→    47→                _ = m.Database.CreateCollection(ctx, cs.Collection)
    48→    48→
    49→    49→                cmd := bson.D{
    50→    50→                        {Key: "collMod", Value: cs.Collection},
    51→    51→                        {Key: "validator", Value: cs.Schema},
    52→    52→                        {Key: "validationLevel", Value: "moderate"},
    53→    53→                        {Key: "validationAction", Value: "error"},
    54→    54→                }
    55→    55→
    56→    56→                if err := m.Database.RunCommand(ctx, cmd).Err(); err != nil {
    57→    57→                        slog.Warn("failed to apply schema validation", "collection", cs.Collection, "error", err)
    58→    58→                }
    59→    59→        }
    60→    60→}
