    10→
    11→// CollectionSchema pairs a collection name with its JSON Schema validator.
    12→type CollectionSchema struct {
    13→        Collection string
    14→        Schema     bson.M
    15→}
    16→
    17→// AllSchemas returns the JSON Schema validators for all 15 validated collections.
    18→func AllSchemas() []CollectionSchema {
    19→        return []CollectionSchema{
    20→                usersSchema(),
    21→                tenantsSchema(),
    22→                tenantMembershipsSchema(),
    23→                invitationsSchema(),
    24→                plansSchema(),
    25→                creditBundlesSchema(),
    26→                financialTransactionsSchema(),
    27→                webhooksSchema(),
    28→                apiKeysSchema(),
    29→                configVarsSchema(),
    30→                announcementsSchema(),
    31→                customPagesSchema(),
    32→                messagesSchema(),
    33→                usageEventsSchema(),
    34→                ssoConnectionsSchema(),
    35→                eventDefinitionsSchema(),
    36→        }
    37→}
    38→
    39→// EnsureSchemaValidation applies JSON Schema validators to all validated
    40→// collections using collMod with moderate validation level.
    41→func (m *MongoDB) EnsureSchemaValidation() {
    42→        ctx, cancel := context.WithTimeout(context.Background(), 60*time.Second)
    43→        defer cancel()
    44→
    45→        for _, cs := range AllSchemas() {
    46→                // Ensure the collection exists (ignore "already exists" errors).
    47→                _ = m.Database.CreateCollection(ctx, cs.Collection)
    48→
    49→                cmd := bson.D{
    50→                        {Key: "collMod", Value: cs.Collection},
    51→                        {Key: "validator", Value: cs.Schema},
    52→                        {Key: "validationLevel", Value: "moderate"},
    53→                        {Key: "validationAction", Value: "error"},
    54→                }
    55→
    56→                if err := m.Database.RunCommand(ctx, cmd).Err(); err != nil {
    57→                        slog.Warn("failed to apply schema validation", "collection", cs.Collection, "error", err)
    58→                }
    59→        }
    60→}
