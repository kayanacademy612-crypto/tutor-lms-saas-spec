    39→    39→// spaHandler serves a single-page application from a static directory.
    40→    40→// For files that exist on disk, it serves them directly. For all other
    41→    41→// paths it serves index.html so the SPA router can handle them.
    42→    42→// When serving index.html, it replaces {{APP_NAME}} with the actual app name
    43→    43→// to prevent a title flicker while JavaScript loads branding data.
    44→    44→type spaHandler struct {
    45→    45→        staticPath string
    46→    46→        indexPath  string
    47→    47→        getAppName func() string
    48→    48→}
    49→    49→
    50→    50→func (h spaHandler) ServeHTTP(w http.ResponseWriter, r *http.Request) {
    51→    51→        // Never serve the SPA for /api/ paths — those should be handled by API routes
    52→    52→        if strings.HasPrefix(r.URL.Path, "/api/") || r.URL.Path == "/api" {
    53→    53→                http.NotFound(w, r)
    54→    54→                return
    55→    55→        }
    56→    56→
    57→    57→        path := filepath.Join(h.staticPath, filepath.Clean(r.URL.Path))
    58→    58→
    59→    59→        fi, err := os.Stat(path)
    60→    60→        if os.IsNotExist(err) || (err == nil && fi.IsDir()) {
    61→    61→                // Serve index.html with no-store so browsers always fetch the latest version.
    62→    62→                // Replace the {{APP_NAME}} placeholder with the real app name so the
    63→    63→                // browser tab shows the correct title immediately, before JS loads.
    64→    64→                w.Header().Set("Cache-Control", "no-store, no-cache, must-revalidate")
    65→    65→                indexFile := filepath.Join(h.staticPath, h.indexPath)
    66→    66→                data, readErr := os.ReadFile(indexFile)
    67→    67→                if readErr != nil {
    68→    68→                        http.Error(w, "Internal Server Error", http.StatusInternalServerError)
    69→    69→                        return
    70→    70→                }
    71→    71→                appName := "App"
    72→    72→                if h.getAppName != nil {
    73→    73→                        if name := h.getAppName(); name != "" {
    74→    74→                                appName = name
    75→    75→                        }
    76→    76→                }
    77→    77→                html := strings.Replace(string(data), "{{APP_NAME}}", appName, 1)
    78→    78→                w.Header().Set("Content-Type", "text/html; charset=utf-8")
    79→    79→                w.Write([]byte(html))
    80→    80→                return
    81→    81→        }
    82→    82→        if err != nil {
    83→    83→                http.Error(w, "Internal Server Error", http.StatusInternalServerError)
    84→    84→                return
    85→    85→        }
    86→    86→
    87→    87→        http.FileServer(http.Dir(h.staticPath)).ServeHTTP(w, r)
    88→    88→}

   779→   779→                slog.Info("Serving frontend", "staticDir", cfg.Frontend.StaticDir)
   780→   780→                spa := spaHandler{
   781→   781→                        staticPath: cfg.Frontend.StaticDir,
   782→   782→                        indexPath:  "index.html",
   783→   783→                        getAppName: func() string {
   784→   784→                                // Check branding config in DB first, fall back to configstore app.name.
   785→   785→                                var bc models.BrandingConfig
   786→   786→                                if err := database.BrandingConfig().FindOne(context.Background(), bson.M{}).Decode(&bc); err == nil && bc.AppName != "" {
   787→   787→                                        return bc.AppName
   788→   788→                                }
   789→   789→                                return cfgStore.Get("app.name")
   790→   790→                        },
   791→   791→                }
   792→   792→                router.PathPrefix("/").Handler(spa)
   793→   793→        }
   794→   794→
   795→   795→        // CORS
   796→   796→        c := cors.New(cors.Options{
   797→   797→                AllowedOrigins:   []string{cfg.Frontend.URL},
   798→   798→                AllowedMethods:   []string{"GET", "POST", "PUT", "PATCH", "DELETE", "OPTIONS"},
   799→   799→                AllowedHeaders:   []string{"Authorization", "Content-Type", "X-Tenant-ID"},
   800→   800→                ExposedHeaders:   []string{"X-Request-ID", "X-API-Version"},
   801→   801→                AllowCredentials: true,
   802→   802→                MaxAge:           86400,
   803→   803→        })
   804→   804→
   805→   805→        // Wrap with recovery + body size limit + security headers + CORS + metrics
   806→   806→        handler := middleware.Recovery(middleware.BodySizeLimit(middleware.SecurityHeaders(c.Handler(metricsCollector.Middleware(router)))))

   810→   810→        srv := &http.Server{

   812→   812→                Handler:           handler,
   813→   813→                ReadHeaderTimeout: 5 * time.Second,
   814→   814→                ReadTimeout:       15 * time.Second,
   815→   815→                WriteTimeout:      15 * time.Second,
   816→   816→                IdleTimeout:       60 * time.Second,
   817→   817→        }
   818→   818→
   819→   819→        // Graceful shutdown
   820→   820→        go func() {
   821→   821→                slog.Info("Server listening", "addr", addr)

   823→   823→                        slog.Error("Server failed", "error", err)
   824→   824→                        os.Exit(1)
   825→   825→                }
   826→   826→        }()

   836→   836→        shutdownCtx, shutdownCancel := context.WithTimeout(context.Background(), 30*time.Second)
   837→   837→        defer shutdownCancel()
   838→   838→
   839→   839→        if err := srv.Shutdown(shutdownCtx); err != nil {
   840→   840→                slog.Error("Server forced shutdown", "error", err)
   841→   841→                os.Exit(1)
   842→   842→        }
   843→   843→        slog.Info("Server stopped")
   844→   844→}
